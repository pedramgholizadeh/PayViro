Webiro Gateway

PHP SDK

یک فایل، بدون وابستگی: sdk/WebiroGateway.php را از گیت‌وی بردارید و در هر دامنه‌ای (PHP 7.4+ با cURL) استفاده کنید.

نصب#

فایل WebiroGateway.php را کپی کنید (مثلاً در lib/ پروژه‌ی خودتان) و در ابتدای اسکریپت‌ها require کنید. مثال‌های آماده هم در sdk/examples/ هست.

boot.php
require __DIR__ . '/lib/WebiroGateway.php';

$gateway = new WebiroGateway(
    'https://webiro.ir/gateway',   // آدرس گیت‌وی
    'wgr_...',                      // API Key این دامنه
    'wgs_...',                      // Secret (برای امضاها)
    20                              // timeout (اختیاری)
);

متدها#

متدکارخروجی / رفتار
requestPayment($fields)ایجاد پرداختآرایه با token و payment_url — خطاها با WebiroGatewayException
redirect($fields)ایجاد + هدایت مستقیمهدر Location می‌فرستد و اسکریپت تمام می‌شود
status($token)استعلام وضعیتآرایه‌ی کامل وضعیت (منبع حقیقت)
verifyRedirect($query)تایید بازگشت کاربرامضای sig را چک می‌کند؛ دستکاری → WebiroSignatureException
readWebhook()دریافت وبهوکامضای هدر را روی بدنه‌ی خام چک می‌کند؛ خروجی event + data

شروع پرداخت#

payment/start.php
try {
    $payment = $gateway->requestPayment([
        'amount'          => 2500000,                 // ریال
        'callback_url'    => 'https://lahze-app.ir/payment/result',
        'order_id'        => 'ORD-101',
        'description'     => 'خرید اشتراک یک‌ساله',
        'mobile'          => '09121234567',
        'gateway'         => 'auto',
        'idempotency_key' => 'ORD-101',              // جلوگیری از تراکنش تکراری
    ]);

    // توکن را کنار سفارش ذخیره کنید تا بعداً استعلام بگیرید
    saveOrderToken('ORD-101', $payment['token']);

    header('Location: ' . $payment['payment_url']);
    exit;

} catch (WebiroGatewayException $e) {
    // $e->errorCode() و $e->httpStatus() برای منطق برنامه
    echo 'خطا: ' . htmlspecialchars($e->getMessage());
}

صفحه‌ی بازگشت#

payment/result.php
try {
    $result = $gateway->verifyRedirect();   // پیش‌فرض: $_GET

    // ✅ توصیه‌شده برای مبالغ بالا: استعلام دوباره از منبع حقیقت
    $status = $gateway->status($result['token']);
    if ($status['status'] !== 'paid') {
        $result['ok'] = false;
    }

    if ($result['ok']) {
        markOrderPaid($result['order_id'], $result['ref_no'], $result['amount']);
        echo 'پرداخت موفق — کد پیگیری: ' . htmlspecialchars($result['ref_no']);
    } else {
        echo 'پرداخت ناموفق';
    }

} catch (WebiroSignatureException $e) {
    http_response_code(400);
    exit('پارامترهای بازگشت دستکاری شده‌اند.');
}

دریافت وبهوک#

api/payment/webhook.php
try {
    $hook = $gateway->readWebhook();
} catch (WebiroSignatureException $e) {
    http_response_code(400);
    exit;
}

$data = $hook['data'];

switch ($hook['event']) {
    case 'payment.succeeded':
        markOrderPaid($data['order_id'], $data['ref_no'], $data['amount']);
        break;
    case 'payment.failed':
    case 'payment.canceled':
        markOrderUnpaid($data['order_id'], $data['status']);
        break;
}

http_response_code(200);   // بدون 2xx گیت‌وی دوباره تلاش می‌کند
echo 'ok';

دو خط دفاع: سفارش را هم با ریدایرکت ثبت کنید، هم با وبهوک؛ چون عملیات ثبت سفارش idempotent نوشته شده (مثل UPDATE … WHERE paid=0)، اجرای دوباره مشکلی ایجاد نمی‌کند.

استفاده در لاراول#

routes/web.php + app/Http/Controllers/PaymentController.php
// فایل SDK را در app/Libraries/ کپی کنید (namespace آن global است)
Route::get('/payment/start/{order}', [PaymentController::class, 'start']);
Route::post('/payment/result', [PaymentController::class, 'result'])->name('payment.result');
Route::post('/payment/webhook', [PaymentController::class, 'webhook'])->withoutMiddleware(VerifyCsrfToken::class);

public function start(Order $order)
{
    $payment = $order->getGatewayTokenAttribute() ?? null;

    $gw = new \WebiroGateway(config('services.webiro.base_url'), config('services.webiro.key'), config('services.webiro.secret'));
    $p = $gw->requestPayment([
        'amount'       => $order->amount_rial,
        'callback_url' => route('payment.result'),
        'order_id'     => $order->id,
        'mobile'       => $order->user->mobile,
    ]);
    $order->update(['gateway_token' => $p['token']]);

    return redirect($p['payment_url']);
}

بدون SDK (cURL خام)#

خام — بدون فایل SDK
$ch = curl_init('https://webiro.ir/gateway/api/v1/payments');
curl_setopt_array($ch, [
    CURLOPT_POST           => true,
    CURLOPT_RETURNTRANSFER => true,
    CURLOPT_HTTPHEADER     => ['X-API-Key: wgr_...', 'Content-Type: application/json'],
    CURLOPT_POSTFIELDS     => json_encode([
        'amount'       => 2500000,
        'callback_url' => 'https://lahze-app.ir/payment/result',
        'order_id'     => 'ORD-101',
    ]),
]);
$res = json_decode(curl_exec($ch), true);
header('Location: ' . $res['data']['payment_url']);

سه قانون طلایی: ۱) مبلغ همیشه از دیتابیس خودتان، نه ورودی کاربر. ۲) ثبت سفارش idempotent باشد (بدون ثبت تکراری). ۳) کلید secret فقط سمت سرور — هرگز در جاوااسکریپت یا گیت عمومی.
